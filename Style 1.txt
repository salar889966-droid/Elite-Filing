"use client";

import { useState, useEffect } from "react";
import Link from "next/link";
import { motion, AnimatePresence } from "framer-motion";
import { normalizeMovie } from "@/lib/providers/utils";
import WatchlistButton from "./WatchlistButton";
import { MessageSquare, Star } from "lucide-react"; // npm install lucide-react

interface FeaturedSpotlightProps {
  movies: any[];
  loading?: boolean;
}

export default function FeaturedSpotlight({ movies, loading = false }: FeaturedSpotlightProps) {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isIntroDone, setIsIntroDone] = useState(false);

  useEffect(() => {
    if (loading || !movies?.length) return;
    const timer = setInterval(() => {
      setCurrentIndex((prev) => (prev + 1) % Math.min(movies.length, 10));
    }, 8000);
    return () => clearInterval(timer);
  }, [loading, movies]);

  useEffect(() => {
    const timer = setTimeout(() => setIsIntroDone(true), 2000);
    return () => clearTimeout(timer);
  }, []);

  if (loading || !isIntroDone) return <SplashLoader />;
  if (!movies || movies.length === 0) return null;

  const currentRawMovie = movies[currentIndex];
  const movie = normalizeMovie(currentRawMovie);

  // Data extraction
  const backdropSrc = movie.backdropUrl || movie.backdrop || movie.backdrop_path;
  const title = movie.title || "Unknown Title";
  const overview = movie.overview || currentRawMovie.overview || "No description available.";
  const year = movie.release_date?.split("-")[0] || movie.year || "N/A";
  const rating = movie.vote_average?.toFixed(1) || "0.0";
  const votes = movie.vote_count?.toLocaleString() || "0";
  const language = (movie.original_language || "EN").toUpperCase();

  return (
    <section className="relative w-full h-[70vh] md:h-[85vh] min-h-[550px] overflow-hidden bg-black rounded-3xl border border-white/5 shadow-2xl">
      {/* Background Image with AnimatePresence */}
      <AnimatePresence mode="wait">
        <motion.div
          key={`bg-${movie.id}`}
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
          transition={{ duration: 1 }}
          className="absolute inset-0"
        >
          {backdropSrc && (
            <img
              src={backdropSrc}
              alt=""
              className="w-full h-full object-cover object-top md:object-center transition-transform duration-[10s] scale-100 hover:scale-110"
            />
          )}
          {/* Gradient Overlays to match the image style */}
          <div className="absolute inset-0 bg-gradient-to-t from-black via-black/40 to-transparent" />
          <div className="absolute inset-0 bg-gradient-to-r from-black/80 via-black/20 to-transparent" />
        </motion.div>
      </AnimatePresence>

      {/* Text Content */}
      <div className="absolute inset-0 z-10 flex flex-col justify-end p-6 md:p-20 pb-24 md:pb-32">
        <AnimatePresence mode="wait">
          <motion.div
            key={`content-${movie.id}`}
            initial={{ opacity: 0, y: 30 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -30 }}
            transition={{ duration: 0.6 }}
            className="max-w-3xl space-y-6"
          >
            {/* Title */}
            <h1 className="text-4xl md:text-7xl font-bold text-white tracking-tight drop-shadow-2xl">
              {title}
            </h1>

            {/* Metadata Badges */}
            <div className="flex flex-wrap items-center gap-3">
              <span className="bg-white/10 backdrop-blur-md border border-white/20 text-white px-3 py-1 rounded font-medium text-sm">
                {year}
              </span>
              <span className="bg-white/10 backdrop-blur-md border border-white/20 text-white px-3 py-1 rounded flex items-center gap-1.5 font-medium text-sm">
                <Star className="w-4 h-4 text-yellow-500 fill-yellow-500" />
                {rating} <span className="opacity-50 font-normal">({votes} votes)</span>
              </span>
              <span className="bg-white/10 backdrop-blur-md border border-white/20 text-white px-3 py-1 rounded font-medium text-sm uppercase">
                {language}
              </span>
            </div>

            {/* Description */}
            <p className="text-gray-200 text-base md:text-lg leading-relaxed max-w-2xl drop-shadow-md line-clamp-3">
              {overview}
            </p>

            {/* Action Buttons */}
            <div className="flex flex-wrap items-center gap-4 pt-4">
              <Link
                href={`/movie/${movie.id}`}
                className="flex items-center gap-2 px-8 py-3.5 bg-brand-red hover:bg-brand-red/90 text-white font-bold rounded-xl transition-all hover:scale-105 active:scale-95 shadow-lg shadow-brand-red/20"
              >
                <MessageSquare className="w-5 h-5" />
                Start a Discussion
              </Link>

              <div className="scale-110 origin-left">
                <WatchlistButton movieId={movie.id} />
              </div>
            </div>
          </motion.div>
        </AnimatePresence>
      </div>

      {/* Progress Dots */}
      <div className="absolute bottom-10 left-6 md:left-20 flex gap-2 z-20">
        {movies.slice(0, 10).map((_, i) => (
          <button
            key={i}
            onClick={() => setCurrentIndex(i)}
            className={`h-1.5 transition-all duration-500 rounded-full ${i === currentIndex ? "w-10 bg-brand-red" : "w-3 bg-white/30"
              }`}
          />
        ))}
      </div>
    </section>
  );
}
